#### 金额输入限制
```typescript
// 格式化金额，只允许数字和小数点
function formatMoney(value: string): string {
  let v = value.replace(/[^\d.]/g, '')
  const parts = v.split('.')
  if (parts.length > 2) {
    v = parts[0] + '.' + parts.slice(1).join('')
  }
  if (parts[1]?.length > 2) {
    v = parts[0] + '.' + parts[1].slice(0, 2)
  }
  return v
}

// 强制更新 input 显示值，防止中文输入
function onMoneyInput(e: Event) {
  const input = e.target as HTMLInputElement
  const formatted = formatMoney(input.value)
  money.value = formatted
  input.value = formatted  // 关键：强制同步
}
```

### 3. 图片压缩工具 (shared/utils/image-compress.ts)

#### 压缩策略
- **最大尺寸**: 1920 x 1920
- **初始质量**: 0.8
- **目标大小**: 2MB
- **递归压缩**: 超过目标大小时降低质量重试

```typescript
export function compressImage(file: File, options: CompressOptions = {}): Promise<File> {
  const { maxWidth = 1920, maxHeight = 1920, quality = 0.8, maxSizeMB = 2 } = options

  return new Promise((resolve, reject) => {
    // 小于目标大小直接返回
    if (file.size <= maxSizeMB * 1024 * 1024) {
      resolve(file)
      return
    }

    const reader = new FileReader()
    reader.onload = (e) => {
      const img = new Image()
      img.onload = () => {
        // 计算压缩后尺寸
        const { width, height } = calculateSize(img.width, img.height, maxWidth, maxHeight)

        // Canvas 绘制并导出
        const canvas = document.createElement('canvas')
        canvas.width = width
        canvas.height = height
        const ctx = canvas.getContext('2d')
        ctx.drawImage(img, 0, 0, width, height)

        canvas.toBlob((blob) => {
          if (blob.size > maxSizeMB * 1024 * 1024 && quality > 0.3) {
            // 递归压缩，降低质量
            compressImage(new File([blob], file.name), { ...options, quality: quality - 0.1 })
              .then(resolve)
          } else {
            resolve(new File([blob], file.name, { type: file.type }))
          }
        }, file.type, quality)
      }
      img.src = e.target.result as string
    }
    reader.readAsDataURL(file)
  })
}
```

## 遇到的问题与解决

### 1. 金额输入允许中文字符
**问题**: 使用中文输入法时，金额字段可以输入汉字
**原因**: Vue 的 v-model 绑定在输入法组合过程中不会实时更新
**解决**: 在 `onInput` 事件中强制更新 `input.value`

```typescript
function onMoneyInput(e: Event) {
  const input = e.target as HTMLInputElement
  const formatted = formatMoney(input.value)
  money.value = formatted
  input.value = formatted  // 强制同步，阻止输入法
}
```

### 2. 图片上传显示错位
**问题**: 上传图片后，loading 状态显示在错误的位置
**原因**: Vant Uploader 会自动添加文件到 fileList，代码又 push 了一次
**解决**: 使用 `detail.index` 找到已有项更新，而非 push 新项

```typescript
function onAfterRead({ file, detail }: { file: any; detail: { index: number } }) {
  const index = detail.index
  if (fileList.value[index]) {
    fileList.value[index].status = 'uploading'
    // 压缩并上传...
  }
}
```

### 3. 返回按钮跳转到登录页
**问题**: 从筛选页返回，跳转到登录页而非明细页
**原因**: `router.back()` 在历史记录不存在时行为不确定
**解决**: 改用 `router.push('/flow')` 固定跳转

### 4. 分类列表不显示
**问题**: `typeApi.getAllWithChildren is not a function`
**原因**: API 方法名错误
**解决**: 改用 `typeApi.getAllNoLimit()`

### 5. 追加分账单布局错乱
**问题**: 备注输入框和删除按钮不可见
**原因**: flex 子元素溢出
**解决**: 添加 `min-width: 0` 和 `flex-shrink: 0`

## 颜色标识系统

资金流向和收支类型使用统一的颜色标识：

| 类型 | 颜色 | CSS 变量 |
|-----|------|----------|
| 收入 | 绿色 | `--color-income` (#52C41A) |
| 支出 | 红色 | `--color-expense` (#FF4D4F) |
| 转账 | 蓝色 | `--color-transfer` (#1890FF) |

```typescript
function getHandleClass(handle: number | undefined): string {
  if (handle === 0) return 'income'   // 收入
  if (handle === 1) return 'expense'  // 支出
  if (handle === 2) return 'transfer' // 转账
  return ''
}
```
