## 今日开发内容

### 1. 备份恢复 API 接口

#### 1.1 新增 `src/shared/api/backup.ts`

```typescript
import { getRequest } from './request'
import type { ApiResponse } from '../types'

export const backupApi = {
  /**
   * 手动备份
   * POST /backup/backup
   * 生成备份文件 → 发送 WebHook（邮件）→ 返回文件名
   */
  backup() {
    return getRequest().post<ApiResponse<string>>('/backup/backup')
  },

  /**
   * 上传恢复
   * POST /backup/restore (multipart/form-data)
   * 上传 SQL 备份文件进行数据恢复
   */
  restore(file: File) {
    const formData = new FormData()
    formData.append('file', file)

    return getRequest().post<ApiResponse<null>>('/backup/restore', formData, {
      headers: {
        'Content-Type': 'multipart/form-data',
      },
      timeout: 120000, // 恢复操作 2 分钟超时
    })
  },
}
```

#### 1.2 API 使用方式

```typescript
import { backupApi } from '@shared/api'

// 手动备份
const res = await backupApi.backup()
console.log(res.data.data) // "yd_jz_manual_20260125_1530.sql"

// 上传恢复
const file = inputElement.files[0]
await backupApi.restore(file)
```

```typescript
// 选择文件
function onFileChange(e: Event) {
  const target = e.target as HTMLInputElement
  const file = target.files?.[0]
  if (file) {
    if (!file.name.endsWith('.sql')) {
      ElMessage.warning('请选择 .sql 格式的备份文件')
      target.value = ''
      return
    }
    selectedFile.value = file
  }
}

// 拖拽上传
function onDrop(e: DragEvent) {
  e.preventDefault()
  const file = e.dataTransfer?.files?.[0]
  if (file) {
    if (!file.name.endsWith('.sql')) {
      ElMessage.warning('请选择 .sql 格式的备份文件')
      return
    }
    selectedFile.value = file
  }
}
```

### 2. 二次确认与恢复流程

```typescript
async function doRestore() {
  // 二次确认
  try {
    await ElMessageBox.prompt(
      '此操作将覆盖当前所有数据，且无法撤销！\n\n请输入"确认恢复"以继续：',
      '确认恢复数据',
      {
        inputPattern: /^确认恢复$/,
        inputErrorMessage: '请输入"确认恢复"',
      }
    )
  } catch {
    return // 用户取消
  }

  // 执行恢复
  const res = await backupApi.restore(selectedFile.value)
  if (res.data.code === 0) {
    // 恢复成功，强制重新登录
    ElMessageBox.alert('数据恢复成功！为确保数据一致性，请重新登录。', '恢复成功', {
      callback: () => {
        localStorage.removeItem('token')
        window.location.href = '/auth?mode=1'
      }
    })
  }
}
```
