```typescript
/**
 * 内部转账豁免模式
 * 仅对内部转账（handle=2）生效，用于控制转账时哪个账户的金额变动不计入净资产
 */
export enum ExemptMode {
  /** 都不豁免（默认） */
  NONE = 0,
  /** 转出账户豁免 */
  FROM_EXEMPT = 1,
  /** 转入账户豁免（如还信用卡） */
  TO_EXEMPT = 2,
  /** 两边都豁免 */
  BOTH_EXEMPT = 3,
}

// Action 和 ActionParams 接口新增字段
exemptMode?: ExemptMode
```

#### 3.2 双向计算属性

使用 computed getter/setter 实现双向绑定：

```typescript
const fromExempt = computed({
  get: () => {
    const mode = actionForm.value.exemptMode ?? ExemptMode.NONE
    return mode === ExemptMode.FROM_EXEMPT || mode === ExemptMode.BOTH_EXEMPT
  },
  set: (val: boolean) => {
    const toExempt = actionForm.value.exemptMode === ExemptMode.TO_EXEMPT
      || actionForm.value.exemptMode === ExemptMode.BOTH_EXEMPT
    if (val && toExempt) {
      actionForm.value.exemptMode = ExemptMode.BOTH_EXEMPT
    } else if (val) {
      actionForm.value.exemptMode = ExemptMode.FROM_EXEMPT
    } else if (toExempt) {
      actionForm.value.exemptMode = ExemptMode.TO_EXEMPT
    } else {
      actionForm.value.exemptMode = ExemptMode.NONE
    }
  }
})
```

### 4. Bug 修复

#### 4.1 分类更新 Bug 修复

**问题**：编辑分类时，后端创建新分类而不是更新现有分类

**原因**：后端 JPA 的 `save()` 方法通过 body 中的 `id` 字段判断操作类型
- `id = 0` 或不传 → 新增
- `id > 0` → 更新

**修复**：
```typescript
// src/shared/api/type.ts - TypeParams 添加 id 字段
export interface TypeParams {
  id?: number  // 更新时必传
  tname: string
  // ...
}

// 移动端 TypeAdd.vue
await typeApi.update(typeId.value, {
  id: typeId.value,  // 添加 id
  tname: tname.value.trim(),
  // ...
})

// 桌面端 TypeManager.vue
await typeApi.update(editingType.value.id, { ...params, id: editingType.value.id })
```

#### 4.2 FlowEditor 面板状态 Bug 修复

**问题1**：快记模板面板状态没有在重新打开时重置，导致编辑已有流水时面板意外显示

**修复**：在 `initData()` 开始时调用 `closeAllPanels()`

```typescript
async function initData() {
```

**问题2**：面板滚动位置没有重置，关闭后再打开仍在之前位置

**修复**：给面板 body 添加 ref，打开时用 `nextTick` 重置 `scrollTop`

```typescript
const accountPanelBody = ref<HTMLElement | null>(null)
const typePanelBody = ref<HTMLElement | null>(null)
const templatePanelBody = ref<HTMLElement | null>(null)

function openAccountPanel(type: 1 | 2) {
  closeAllPanels()
  accountPanelType.value = type
  accountPanelVisible.value = true
  nextTick(() => {
    if (accountPanelBody.value) accountPanelBody.value.scrollTop = 0
  })
}
```

### 7. ActionManager 列表显示优化

#### 7.1 不计入模式文本

根据 action 类型和 exemptMode 显示不同文本：

```typescript
function getExemptText(action: Action): string {
  if (!action.exempt) return ''
  // 收入/支出只显示"不计入"
  if (action.handle !== ActionHandle.TRANSFER) return '不计入'
  // 内部转账根据模式显示
  switch (action.exemptMode) {
    case ExemptMode.FROM_EXEMPT: return '转出不计入'
    case ExemptMode.TO_EXEMPT: return '转入不计入'
    case ExemptMode.BOTH_EXEMPT: return '两边不计入'
    default: return '不计入'
  }
}
```

**显示效果**：

| 类型 | exemptMode | 显示 |
|------|------------|------|
| 收入/支出 | - | "不计入" |
| 内部转账 | FROM_EXEMPT | "转出不计入" |
| 内部转账 | TO_EXEMPT | "转入不计入" |
| 内部转账 | BOTH_EXEMPT | "两边不计入" |
